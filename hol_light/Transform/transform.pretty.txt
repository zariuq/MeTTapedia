core.ml — Definition tc_enabled
|- forall taken read made present.
       tc_enabled present taken read made <=>
       (forall occ. taken occ ==> present occ) /\
       (forall occ. read occ ==> present occ) /\
       (forall occ. ~(taken occ /\ read occ)) /\
       (forall occ. made occ ==> ~present occ)

core.ml — Definition tc_fire
|- forall present taken made occ.
       tc_fire present taken made occ <=>
       present occ /\ ~taken occ \/ made occ

core.ml — Definition tc_models
|- forall hom a store eq b act sigma.
       tc_models hom eq act a b sigma store <=>
       hom a b sigma /\
       (forall x y. store x y ==> eq b (act sigma x) (act sigma y))

core.ml — Definition tc_solvable
|- forall hom eq act a store.
       tc_solvable hom eq act a store <=>
       (exists b sigma. tc_models hom eq act a b sigma store)

core.ml — Definition tc_entails
|- forall hom eq act a store holds.
       tc_entails hom eq act a store holds <=>
       (forall b sigma.
            tc_models hom eq act a b sigma store ==> holds b sigma)

core.ml — Definition tc_map_store
|- forall store u v act e.
       tc_map_store act e store u v <=>
       (exists x y. store x y /\ u = act e x /\ v = act e y)

core.ml — Definition tc_separates
|- forall apart eq b act sigma.
       tc_separates eq act b sigma apart <=>
       (forall x y. apart x y ==> ~eq b (act sigma x) (act sigma y))

core.ml — Definition tc_solution
|- forall hom a store eq act b sigma apart.
       tc_solution hom eq act a store apart b sigma <=>
       tc_models hom eq act a b sigma store /\
       tc_separates eq act b sigma apart

core.ml — Definition tc_preparable
|- forall ready solution.
       tc_preparable ready solution <=> ready /\ (exists s. solution s)

core.ml — Definition tc_admits
|- forall ready solution holds.
       tc_admits ready solution holds <=>
       tc_preparable ready solution /\ (forall s. solution s ==> holds s)

core.ml — Definition tc_faithful
|- forall terms c act j eq b.
       tc_faithful terms eq act b c j <=>
       (forall x y.
            terms b x /\ terms b y /\ eq c (act j x) (act j y) ==> eq b x y)

core.ml — Definition tc_covers
|- forall store hom r settled terms a eq act e.
       tc_covers terms hom eq act a r e store settled <=>
       (forall b sigma.
            tc_models hom eq act a b sigma store
            ==> (exists c j tau.
                     hom b c j /\
                     tc_faithful terms eq act b c j /\
                     tc_models hom eq act r c tau settled /\
                     (forall t.
                          terms a t
                          ==> eq c (act j (act sigma t)) (act tau (act e t)))))

core.ml — Definition tc_same_target_covers
|- forall store hom r settled terms a eq act e.
       tc_same_target_covers terms hom eq act a r e store settled <=>
       (forall b sigma.
            tc_models hom eq act a b sigma store
            ==> (exists tau.
                     tc_models hom eq act r b tau settled /\
                     (forall t.
                          terms a t
                          ==> eq b (act sigma t) (act tau (act e t)))))

core.ml — Definition tc_covering_family
|- forall source branch recovery.
       tc_covering_family source branch recovery <=>
       (forall i r. branch i r ==> (exists s. recovery s i r)) /\
       (forall i r s. branch i r /\ recovery s i r ==> source s) /\
       (forall s. source s ==> (exists i r. branch i r /\ recovery s i r))

core.ml — Definition tc_prepares

frames.ml — Definition tc_frames
|- forall objects identity comp hom terms eq act.
       tc_frames objects terms hom eq act identity comp <=>
       (forall a t. terms a t ==> objects a) /\
       (forall a b f. hom a b f ==> objects a /\ objects b) /\
       (forall a. objects a ==> hom a a (identity a)) /\
       (forall a b c f g. hom a b f /\ hom b c g ==> hom a c (comp g f)) /\
       (forall a b f t. hom a b f /\ terms a t ==> terms b (act f t)) /\
       (forall a t. terms a t ==> act (identity a) t = t) /\
       (forall a b c f g t.
            hom a b f /\ hom b c g /\ terms a t
            ==> act (comp g f) t = act g (act f t)) /\
       (forall a t. terms a t ==> eq a t t) /\
       (forall a x y. terms a x /\ terms a y /\ eq a x y ==> eq a y x) /\
       (forall a x y z.
            terms a x /\ terms a y /\ terms a z /\ eq a x y /\ eq a y z
            ==> eq a x z) /\
       (forall a b f x y.
            hom a b f /\ terms a x /\ terms a y /\ eq a x y
            ==> eq b (act f x) (act f y))

       ==> (forall events s t.
                tc_path step events s t
                ==> tc_path target (MAP label events) (encode s) (encode t))

runs.ml — Theorem TC_BISIMULATION_SYMMETRIC
|- forall step target relation.
       tc_bisimulation step target relation
       ==> tc_bisimulation target step (\u s. relation s u)

runs.ml — Theorem TC_PATH_BISIMULATION
|- forall step target relation.
       tc_bisimulation step target relation
       ==> (forall events s u t.
                relation s u /\ tc_path step events s t
                ==> (exists v. tc_path target events u v /\ relation t v))

runs.ml — Theorem TC_PATH_BISIMULATION_BACK
|- forall step target relation.
       tc_bisimulation step target relation
       ==> (forall events s u v.
                relation s u /\ tc_path target events u v
                ==> (exists t. tc_path step events s t /\ relation t v))

runs.ml — Theorem TC_BISIMULATION_OBSERVATIONS
|- forall step target relation left right.
       tc_bisimulation step target relation /\
       (forall s u. relation s u ==> left s = right u)
       ==> (forall events s u answer.
                relation s u
                ==> (tc_observes step left events s answer <=>
                     tc_observes target right events u answer))

runs.ml — Theorem TC_PATH_SWAP
|- forall step first second.
       (forall s u t.
            step s first u /\ step u second t
            ==> (exists v. step s second v /\ step v first t))
       ==> (forall rest s t.
                tc_path step (CONS first (CONS second rest)) s t
                ==> tc_path step (CONS second (CONS first rest)) s t)

runs.ml — Theorem TC_POSITIVE_FOOTPRINT_DIAMOND
|- forall present take1 read1 make1 take2 read2 make2.
       tc_enabled present take1 read1 make1 /\
       tc_enabled present take2 read2 make2 /\
       (forall occ. take1 occ ==> ~take2 occ /\ ~read2 occ) /\
       (forall occ. take2 occ ==> ~read1 occ) /\
       (forall occ. make1 occ ==> ~make2 occ)
       ==> tc_enabled (tc_fire present take1 make1) take2 read2 make2 /\
           tc_enabled (tc_fire present take2 make2) take1 read1 make1 /\
           (forall occ.
                tc_fire (tc_fire present take1 make1) take2 make2 occ <=>
                tc_fire (tc_fire present take2 make2) take1 make1 occ)

runs.ml — Definition tc_absent
|- forall present region.
       tc_absent present region <=>
       (forall occ. ~(present occ /\ region occ))

runs.ml — Theorem TC_ABSENCE_NOT_FRAMED
|- tc_absent (\occ. false) (\occ. true) /\
   ~tc_absent (\occ. ~occ) (\occ. true)

runs.ml — Theorem TC_NONTRIVIAL_TWO_EVENT_PATH
|- tc_path (\s event t. t <=> (if event then ~s else s)) [false; true] false
   true /\
   ~tc_path (\s event t. t <=> (if event then ~s else s)) [false; true] false
    false

runs.ml — Theorem TC_FORWARD_SIMULATION_NOT_REFLECTION
|- tc_simulates (\s event t. false) (\s event t. true) (\s. s) (\e. e) /\
   tc_path (\s event t. true) [false] false true /\
   ~tc_path (\s event t. false) [false] false true

runs.ml — Theorem TC_OBSERVER_CAN_FORGET_ENDPOINT
|- tc_observes (\s event t. true) (\s. false) [false] false false /\
   tc_path (\s event t. true) [false] false false /\
   tc_path (\s event t. true) [false] false true
